function doPost(e) { try { var data = JSON.parse(e.postData.contents); var action = data.action; if (action === "generateMonthlyPDF") { return generateMonthlyPDFFile(data.month, data.reportsMap, data.parameters, data.timeSlots); } else if (action === "get_pdf_list") { return getPdfList(); } return ContentService.createTextOutput(JSON.stringify({status: "error", message: "Action tidak dikenal"})) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()})) .setMimeType(ContentService.MimeType.JSON); } } function generateMonthlyPDFFile(monthStr, reportsMap, parameters, timeSlots) { // HTML Template Rekap Bulanan Landscape - Mode Rata-Rata per Tanggal var html = ''; html += '

REKAP BULANAN RATA-RATA DAILY REPORT METERING

'; html += '

Periode Bulan: ' + monthStr + ' | STA. MNCGROUP PALANGKA RAYA

'; // Hitung jumlah hari dalam bulan tersebut var [year, month] = monthStr.split('-').map(Number); var daysInMonth = new Date(year, month, 0).getDate(); var datesList = []; for (var d = 1; d <= daysInMonth; d++) { var dayStr = (d < 10 ? '0' : '') + d; datesList.push(monthStr + '-' + dayStr); } html += ''; html += ''; html += ''; // Header Tanggal (hanya menampilkan tanggalnya saja) datesList.forEach(function(date) { html += ''; }); html += ''; // Baris Data Parameter (Menghitung Rata-rata dari timeSlots yang ada) parameters.forEach(function(cat) { var totalCols = 2 + datesList.length; html += ''; cat.items.forEach(function(param) { html += ''; html += ''; html += ''; datesList.forEach(function(date) { var dayReport = reportsMap[date] || {}; var paramValues = dayReport.meteringData?.[param.id] || {}; // Hitung rata-rata dari nilai-nilai jam yang terisi angka var sum = 0; var count = 0; timeSlots.forEach(function(slot) { var val = paramValues[slot]; if (val !== undefined && val !== null && val !== '' && !isNaN(val)) { sum += Number(val); count++; } }); var avgVal = count > 0 ? (sum / count).toFixed(1) : ''; html += ''; }); html += ''; }); }); html += '
ParameterSatuan' + date.substring(8, 10) + '
' + cat.category + '
' + param.name + '' + param.unit + '' + avgVal + '
'; // Konversi HTML ke PDF Landscape via Google Drive Converter var blob = Utilities.newBlob(html, 'text/html', 'rekap_avg_' + monthStr + '.html'); var file = DriveApp.createFile(blob); var pdfBlob = file.getAs('application/pdf'); var pdfName = 'Rekap_RataRata_Metering_' + monthStr + '.pdf'; var pdfFile = DriveApp.createFile(pdfBlob).setName(pdfName); // Hapus file HTML sementara file.setTrashed(true); return ContentService.createTextOutput(JSON.stringify({ status: "success", fileUrl: pdfFile.getUrl(), fileName: pdfName })).setMimeType(ContentService.MimeType.JSON); } function getPdfList() { var files = DriveApp.getFiles(); var pdfs = []; while (files.hasNext()) { var file = files.next(); if (file.getMimeType() === 'application/pdf' && file.getName().indexOf('Rekap_') === 0) { pdfs.push({ name: file.getName(), url: file.getUrl(), dateCreated: Utilities.formatDate(file.getDateCreated(), "GMT+7", "yyyy-MM-dd HH:mm:ss") }); } } return ContentService.createTextOutput(JSON.stringify({ status: "success", pdfs: pdfs })).setMimeType(ContentService.MimeType.JSON); }
